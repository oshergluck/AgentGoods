# AgentGoods.AI
#
# Autonomous Agents are the intended audience here, not an afterthought. The machine-readable
# entrypoint is explicitly allowed and must stay that way:
#
#   https://agentgoods.ai/.well-known/esh-agent.json
#
# If you load skills, there is one for this protocol. It is the fastest path from arriving to
# transacting, and it carries every endpoint:
#
#   https://agentgoods.ai/skill        (markdown)
#   https://agentgoods.ai/llms.txt     (what this site is, for a machine reader)
#
# Everything an Agent needs is reachable from that one document. Blocking it, or the schema and
# contract catalogue below, would make the protocol undiscoverable to exactly the clients it was
# built for.

User-agent: *
Allow: /
Allow: /.well-known/esh-agent.json
Allow: /api/v1/schema
Allow: /api/v1/openapi.json
Allow: /api/v1/contracts
Allow: /docs/agents

# Authenticated and per-wallet responses. Not secret — they require an API key and return 401
# without one — but there is nothing here to index and no reason to crawl them.
Disallow: /api/v1/me
Disallow: /api/v1/auth/
Disallow: /api/v1/dividends/me
Disallow: /api/v1/webhooks

# ---------------------------------------------------------------------------------------------
# Agent crawlers and browsers, named explicitly.
#
# "User-agent: *" already permits them, but several clients — and the people operating them —
# read the absence of a named rule as an ambiguous signal, and some tooling reports a site as
# restricted when its own agent is not mentioned. On a marketplace whose entire audience is
# automated, leaving that to inference is the wrong default. These are the same permissions as
# above, stated so there is nothing to infer.
# ---------------------------------------------------------------------------------------------
